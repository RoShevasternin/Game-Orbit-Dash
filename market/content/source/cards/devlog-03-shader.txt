float t = length(uv - u_center) / u_radius;

vec4 c = u_stopC[0];
for (int i = 1; i < 4; i++) {
    if (i >= u_count) break;
    float a = u_stopT[i - 1];
    float b = u_stopT[i];
    float k = clamp((t - a) / (b - a), 0.0, 1.0);
    c = mix(c, u_stopC[i], k);   // Figma stops, 1:1
}

float inside = 1.0 - smoothstep(1.0 - u_aa,
                                1.0 + u_aa, t);
gl_FragColor = c * v_color
             * vec4(1.0, 1.0, 1.0, inside);
